# Dynamic QR Code Implementation Plan

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** Memungkinkan pengguna membuat QR code yang URL tujuannya bisa diubah setelah QR code dicetak, serta melacak jumlah scan.

**Architecture:** Backend API untuk menyimpan mapping short URL dan mengelola redirect. Frontend UI untuk CRUD Dynamic QR Code.

**Tech Stack:** Next.js, MongoDB Atlas.

---

## Current Context / Assumptions

- QRkuy saat ini hanya mendukung static QR codes.
- Pengguna membutuhkan kemampuan untuk mengubah destinasi URL tanpa mencetak ulang QR code.
- Pelacakan dasar (jumlah scan) dibutuhkan.
- Next.js API Routes akan digunakan untuk backend.
- MongoDB Atlas akan digunakan sebagai database.

## Proposed Approach

1.  **Database Schema Design:** Buat skema Mongoose untuk `DynamicQr` yang mencakup `short_code`, `target_url`, `scan_count`, `owner_id`, `created_at`, `updated_at`.
2.  **Backend API Routes:**
    *   `POST /api/dynamic-qr`: Membuat entri Dynamic QR baru dan menghasilkan `short_code` unik.
    *   `PUT /api/dynamic-qr/:id`: Memperbarui `target_url` untuk Dynamic QR yang ada.
    *   `GET /api/dynamic-qr/:id`: Mengambil detail Dynamic QR.
    *   `DELETE /api/dynamic-qr/:id`: Menghapus Dynamic QR.
    *   `GET /<short_code>`: Route untuk melakukan redirect dari `short_code` ke `target_url` dan menambah `scan_count`. (Ini mungkin bukan `api` route, tapi route utama Next.js).
3.  **Frontend UI Integration:**
    *   Halaman "My Dynamic QRs" untuk melihat daftar Dynamic QR yang dibuat pengguna.
    *   Form untuk membuat Dynamic QR baru.
    *   Antarmuka untuk mengedit `target_url` dan melihat statistik (`scan_count`).
    *   Tampilkan QR code untuk `short_code` yang bisa didownload.
    *   Autentikasi pengguna untuk memastikan hanya pemilik yang bisa mengelola QR code-nya.

```javascript
// src/lib/dbConnect.js
import mongoose from 'mongoose';

const MONGODB_URI = process.env.MONGODB_URI;

if (!MONGODB_URI) {
  throw new Error(
    'Please define the MONGODB_URI environment variable inside .env.local'
  );
}

let cached = global.mongoose;

if (!cached) {
  cached = global.mongoose = { conn: null, promise: null };
}

async function dbConnect() {
  if (cached.conn) {
    return cached.conn;
  }

  if (!cached.promise) {
    const opts = {
      bufferCommands: false,
    };

    cached.promise = mongoose.connect(MONGODB_URI, opts).then((mongoose) => {
      return mongoose;
    });
  }
  cached.conn = await cached.promise;
  return cached.conn;
}

export default dbConnect;
```

```javascript
// pages/api/dynamic-qr/index.js
import dbConnect from '../../../src/lib/dbConnect';
import DynamicQr from '../../../src/models/DynamicQr';
import { getSession } from 'next-auth/react'; // Assuming NextAuth.js for user session

export default async function handler(req, res) {
  const { method } = req;

  await dbConnect();

  const session = await getSession({ req });
  if (!session) {
    return res.status(401).json({ success: false, message: 'Authentication required' });
  }

  switch (method) {
    case 'POST':
      try {
        const { target_url } = req.body;
        // Generate a short_code. Could be a random string, or based on a counter.
        // For simplicity, let's use a random alphanumeric string for now.
        const short_code = Math.random().toString(36).substring(2, 8); // e.g., "ajh3k2"

        const dynamicQr = await DynamicQr.create({
          short_code,
          target_url,
          owner_id: session.user.id, // Assuming session.user.id holds the user's ID
        });

        res.status(201).json({ success: true, data: dynamicQr });
      } catch (error) {
        console.error(error);
        res.status(400).json({ success: false, message: error.message });
      }
      break;
    case 'GET': // Optionally, list all dynamic QRs for the current user
        try {
            const dynamicQrs = await DynamicQr.find({ owner_id: session.user.id });
            res.status(200).json({ success: true, data: dynamicQrs });
        } catch (error) {
            console.error(error);
            res.status(400).json({ success: false, message: error.message });
        }
        break;
    default:
      res.status(405).json({ success: false, message: `Method ${method} Not Allowed` });
      break;
  }
}
```

```javascript
import dbConnect from '../src/lib/dbConnect';
import DynamicQr from '../src/models/DynamicQr';

export async function getServerSideProps(context) {
  await dbConnect();
  const { shortCode } = context.query;

  try {
    const dynamicQr = await DynamicQr.findOneAndUpdate(
      { short_code: shortCode },
      { $inc: { scan_count: 1 } },
      { new: true }

        redirect: {

        notFound: true,
      };

    console.error(`Error redirecting for shortCode ${shortCode}:`, error);
    return {
      notFound: true,
    };
  }
}
```

**Step 1: Buat API route `pages/api/dynamic-qr/[id].js`**

```javascript
// pages/api/dynamic-qr/[id].js
import dbConnect from '../../../src/lib/dbConnect';
import DynamicQr from '../../../src/models/DynamicQr';
import { getSession } from 'next-auth/react';

export default async function handler(req, res) {
  const {
    query: { id },
    method,
  } = req;

  await dbConnect();

  const session = await getSession({ req });
  if (!session) {
    return res.status(401).json({ success: false, message: 'Authentication required' });
  }

  switch (method) {
    case 'GET': // Get single Dynamic QR details
      try {
        const dynamicQr = await DynamicQr.findOne({ _id: id, owner_id: session.user.id });
        if (!dynamicQr) {
          return res.status(404).json({ success: false, message: 'Dynamic QR not found or unauthorized' });
        }
        res.status(200).json({ success: true, data: dynamicQr });
      } catch (error) {
        console.error(error);
        res.status(400).json({ success: false, message: error.message });
      }
      break;

    case 'PUT':
      try {
        const { target_url } = req.body;
        const dynamicQr = await DynamicQr.findOneAndUpdate(
          { _id: id, owner_id: session.user.id }, // Find by ID and owner for security
          { target_url, updated_at: new Date() },
          { new: true, runValidators: true } // Return the updated document, run schema validators
        );

        if (!dynamicQr) {
          return res.status(404).json({ success: false, message: 'Dynamic QR not found or unauthorized' });
        }
        res.status(200).json({ success: true, data: dynamicQr });
      } catch (error) {
        console.error(error);
        res.status(400).json({ success: false, message: error.message });
      }
      break;

    case 'DELETE':
      try {
        const deletedDynamicQr = await DynamicQr.deleteOne({ _id: id, owner_id: session.user.id });
        if (deletedDynamicQr.deletedCount === 0) {
          return res.status(404).json({ success: false, message: 'Dynamic QR not found or unauthorized' });
        }
        res.status(200).json({ success: true, data: {} });
      } catch (error) {
        console.error(error);
        res.status(400).json({ success: false, message: error.message });
      }
      break;

    default:
      res.status(405).json({ success: false, message: `Method ${method} Not Allowed` });
      break;
  }
}
```

```javascript
// pages/my-dynamic-qrs.js
import { useState, useEffect } from 'react';
import { useSession } from 'next-auth/react';
import Head from 'next/head';
import DynamicQrForm from '../components/DynamicQrForm';
import DynamicQrCard from '../components/DynamicQrCard'; // Akan dibuat nanti

export default function MyDynamicQRs() {
  const { data: session, status } = useSession();
  const [dynamicQrs, setDynamicQrs] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    if (status === 'authenticated') {
      fetchDynamicQrs();
    } else if (status === 'unauthenticated') {
      setLoading(false);
      // Redirect to login or show message
    }
  }, [status]);

  const fetchDynamicQrs = async () => {
    setLoading(true);
    setError(null);
    try {
      const res = await fetch('/api/dynamic-qr');
      const data = await res.json();
      if (data.success) {
        setDynamicQrs(data.data);
      } else {
        setError(data.message || 'Failed to fetch dynamic QRs');
      }
    } catch (err) {
      setError('An error occurred while fetching dynamic QRs.');
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  const handleQrCreated = (newQr) => {
    setDynamicQrs([...dynamicQrs, newQr]);
  };

  const handleQrUpdated = (updatedQr) => {
    setDynamicQrs(dynamicQrs.map(qr => qr._id === updatedQr._id ? updatedQr : qr));
  };

  const handleQrDeleted = (deletedId) => {
    setDynamicQrs(dynamicQrs.filter(qr => qr._id !== deletedId));
  };

  if (status === 'loading' || loading) {
    return <p>Loading...</p>;
  }

  if (status === 'unauthenticated') {
    return <p>You need to be logged in to view this page.</p>; // Or redirect to login
  }

  return (
    <div className="container mx-auto p-4">
      <Head>
        <title>My Dynamic QR Codes - QRkuy</title>
      </Head>
      <h1 className="text-3xl font-bold mb-6">My Dynamic QR Codes</h1>

      <DynamicQrForm onQrCreated={handleQrCreated} />

      {error && <p className="text-red-500 mb-4">{error}</p>}

      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6 mt-8">
        {dynamicQrs.length > 0 ? (
          dynamicQrs.map((qr) => (
            <DynamicQrCard
              key={qr._id}
              qr={qr}
              onQrUpdated={handleQrUpdated}
              onQrDeleted={handleQrDeleted}
            />
          ))
        ) : (
          <p className="text-gray-600">No dynamic QR codes yet. Create one above!</p>
        )}
      </div>
    </div>
  );
}
```

```javascript
// components/DynamicQrForm.js
import { useState } from 'react';

export default function DynamicQrForm({ onQrCreated }) {
  const [targetUrl, setTargetUrl] = useState('');
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState(null);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setIsSubmitting(true);
    setError(null);

    try {
      const res = await fetch('/api/dynamic-qr', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ target_url: targetUrl }),
      });
      const data = await res.json();

      if (data.success) {
        onQrCreated(data.data);
        setTargetUrl('');
      } else {
        setError(data.message || 'Failed to create Dynamic QR');
      }
    } catch (err) {
      setError('An error occurred while creating Dynamic QR.');
      console.error(err);
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <div className="bg-white p-6 rounded-lg shadow-md mb-8">
      <h2 className="text-2xl font-semibold mb-4">Create New Dynamic QR</h2>
      <form onSubmit={handleSubmit} className="flex flex-col space-y-4">
```

```javascript
// components/DynamicQrCard.js
import { useState } from 'react';
import QRCode from 'qrcode.react'; // Anda perlu menginstal 'qrcode.react' (npm install qrcode.react)

export default function DynamicQrCard({ qr, onQrUpdated, onQrDeleted }) {
  const [isEditing, setIsEditing] = useState(false);
  const [newTargetUrl, setNewTargetUrl] = useState(qr.target_url);
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [error, setError] = useState(null);

  const handleUpdate = async () => {
    setIsSubmitting(true);
    setError(null);
    try {
      const res = await fetch(`/api/dynamic-qr/${qr._id}`, {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ target_url: newTargetUrl }),
      });
      const data = await res.json();
      if (data.success) {
        onQrUpdated(data.data);
        setIsEditing(false);
      } else {
        setError(data.message || 'Failed to update Dynamic QR');
      }
    } catch (err) {
      setError('An error occurred while updating Dynamic QR.');
      console.error(err);
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleDelete = async () => {
    if (!confirm('Are you sure you want to delete this Dynamic QR?')) return;

    setIsSubmitting(true);
    setError(null);
    try {
      const res = await fetch(`/api/dynamic-qr/${qr._id}`, {
        method: 'DELETE',
      });
      const data = await res.json();
      if (data.success) {
        onQrDeleted(qr._id);
      } else {
        setError(data.message || 'Failed to delete Dynamic QR');
      }
    } catch (err) {
      setError('An error occurred while deleting Dynamic QR.');
      console.error(err);
    } finally {
      setIsSubmitting(false);
    }
  };

  // URL yang akan di-encode ke QR Code
  // Asumsi base URL adalah environment variable atau hardcoded 'http://localhost:3000/'
  const qrCodeValue = `${window.location.origin}/${qr.short_code}`;

  return (
    <div className="bg-white p-6 rounded-lg shadow-md flex flex-col items-center">
      <h3 className="text-xl font-semibold mb-2">Short Code: {qr.short_code}</h3>
      <div className="mb-4">
        {qrCodeValue && <QRCode value={qrCodeValue} size={128} level="H" />}
      </div>
      <p className="text-gray-700 break-all mb-2">
        Current URL: {qr.target_url}
      </p>
      <p className="text-gray-700 text-sm mb-4">Scans: {qr.scan_count}</p>

      {isEditing ? (
        <div className="w-full">
          <input
            type="url"
            value={newTargetUrl}
            onChange={(e) => setNewTargetUrl(e.target.value)}
            className="mt-1 block w-full border border-gray-300 rounded-md shadow-sm p-2 mb-2"
            required
          />
          {error && <p className="text-red-500 text-sm mb-2">{error}</p>}
          <div className="flex justify-between gap-2">
            <button
              onClick={handleUpdate}
              className="bg-green-600 text-white py-2 px-4 rounded-md hover:bg-green-700 flex-1"
              disabled={isSubmitting}
            >
              {isSubmitting ? 'Updating...' : 'Save'}
            </button>
            <button
              onClick={() => setIsEditing(false)}
              className="bg-gray-400 text-white py-2 px-4 rounded-md hover:bg-gray-500 flex-1"
              disabled={isSubmitting}
            >
              Cancel
            </button>
          </div>
        </div>
      ) : (
        <div className="flex justify-between gap-2 w-full">
          <button
            onClick={() => setIsEditing(true)}
            className="bg-blue-600 text-white py-2 px-4 rounded-md hover:bg-blue-700 flex-1"
          >
            Edit URL
          </button>
          <button
            onClick={handleDelete}
            className="bg-red-600 text-white py-2 px-4 rounded-md hover:bg-red-700 flex-1"
            disabled={isSubmitting}
          >
            {isSubmitting ? 'Deleting...' : 'Delete'}
          </button>
        </div>
      )}
    </div>
  );
}
```

## Risks, Tradeoffs, and Open Questions

-   **Short Code Collision:** Saat ini menggunakan `Math.random().toString(36).substring(2, 8)`. Risiko kolisi rendah tapi ada. Perlu mekanisme retry atau generator yang lebih canggih jika terjadi kolisi.
-   **Security (Owner ID):** Memastikan bahwa hanya `owner_id` yang benar yang dapat memodifikasi atau menghapus Dynamic QR. Sudah diimplementasikan di API routes.
-   **Analitik Lanjut:** Saat ini hanya `scan_count`. Potensi untuk menambahkan data geo-lokasi, waktu scan, perangkat, dll. Ini memerlukan penambahan skema database dan UI.
-   **Hosting Short URL:** QR code akan mengarah ke domain aplikasi ini (e.g., `qrkuy.com/short_code`).
-   **Rate Limiting:** Menerapkan rate limiting pada API untuk mencegah penyalahgunaan.
